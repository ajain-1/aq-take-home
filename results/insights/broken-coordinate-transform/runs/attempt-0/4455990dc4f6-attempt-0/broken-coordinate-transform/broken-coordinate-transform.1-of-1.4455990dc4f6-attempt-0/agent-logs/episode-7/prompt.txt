
    int A[r1][c1], B[r2][c2], C[r1][c2];

    for(int i=0; i<r1; i++) {
        for(int j=0; j<c1; j++) {
            cin >> A[i][j];
        }
    }

    for(int i=0; i<r2; i++) {
        for(int j=0; j<c2; j++) {
            cin >> B[i][j];
        }
    }

    if(c1 != r2) {
        cout << "Matrix multiplication not possible" << endl;
        return 0;
    }

    for(int i=0; i<r1; i++) {
        for(int j=0; j<c2; j++) {
            C[i][j] = 0;
            for(int k=0; k<c1; k++) {
                C[i][j] += A[i][k] * B[k][j];
            }
        }
    }

    for(int i=0; i<r1; i++) {
        for(int j=0; j<c2; j++) {
            if(j > 0) cout << " ";
            cout << C[i][j];
        }
        cout << endl;
    }

    return 0;
}
