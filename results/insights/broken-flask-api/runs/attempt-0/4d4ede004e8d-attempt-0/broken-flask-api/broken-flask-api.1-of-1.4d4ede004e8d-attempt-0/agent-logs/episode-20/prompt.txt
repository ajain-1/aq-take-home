    try:
        user = next((u for u in users_data if u['id'] == int(user_id)), None)
    except ValueError:
        return jsonify({'error': 'User not found'}), 404

    if user:
        return jsonify(user)
    else:
        return jsonify({'error': 'User not found'}), 404
