import sys
import json
# naive: assumes comma delimiter, no header, outputs list-of-lists

def main():
    if len(sys.argv) != 2 or sys.argv[1].startswith("-"):
        print("usage: python csv_to_json.py <input.csv>")
        sys.exit(2)

    path = sys.argv[1]
    rows = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(",")  # BUG: hard-coded delimiter
            rows.append(parts)

    # BUG: ignores header; emits lists instead of dicts; no pretty option
    print(json.dumps(rows))

if __name__ == "__main__":
    main()

import sys
import os
import json

def main():
    # Parse arguments
    args = sys.argv[1:]
    pretty = False
    path = None

    for arg in args:
        if arg == '--pretty':
            pretty = True
        elif not arg.startswith('-'):
            path = arg

    if path is None:
        print("usage: python csv_to_json.py [--pretty] <input.csv>", file=sys.stderr)
        sys.exit(2)

    # Get delimiter from environment variable, default to comma
    delimiter = os.environ.get('DELIM', ',')

    rows = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            parts = line.split(delimiter)
            rows.append(parts)

    if not rows:
        result = []
    else:
        # First row is header
        header = rows[0]
        data_rows = rows[1:]

        # Convert to list of dicts
        result = []
        for row in data_rows:
            obj = {}
            for i, key in enumerate(header):
                if i < len(row):
                    obj[key] = row[i]
                else:
                    obj[key] = ''
            result.append(obj)

    # Output JSON
    if pretty:
        output = json.dumps(result, ensure_ascii=False, indent=2)
    else:
        output = json.dumps(result, ensure_ascii=False)

    # Write to stdout with UTF-8 encoding
    sys.stdout.write(output + '\n')

if __name__ == "__main__":
    main()
