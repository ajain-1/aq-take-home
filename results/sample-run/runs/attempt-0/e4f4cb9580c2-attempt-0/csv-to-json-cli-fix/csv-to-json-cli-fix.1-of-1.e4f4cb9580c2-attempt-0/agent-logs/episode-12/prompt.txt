import sys
import os
import json

def main():
    # Parse arguments
    args = sys.argv[1:]
    pretty = False
    path = None

    for arg in args:
        if arg == '--pretty':
            pretty = True
        elif not arg.startswith('-'):
            path = arg

    if path is None:
        print("usage: python csv_to_json.py [--pretty] <input.csv>", file=sys.stderr)
        sys.exit(2)
